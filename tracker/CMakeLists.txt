cmake_minimum_required(VERSION 3.16)
project(tracker LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

option(TRK_TSAN "ThreadSanitizer" OFF)
option(TRK_ASAN "AddressSanitizer + UndefinedBehaviorSanitizer" OFF)
option(TRK_WERROR "Warnings are errors" ON)
if(TRK_TSAN AND TRK_ASAN)
  message(FATAL_ERROR "TRK_TSAN and TRK_ASAN cannot be combined")
endif()
if(TRK_TSAN)
  add_compile_options(-fsanitize=thread -g -O1)
  add_link_options(-fsanitize=thread)
endif()
if(TRK_ASAN)
  add_compile_options(-fsanitize=address,undefined -fno-sanitize-recover=undefined -fno-omit-frame-pointer -g -O1)
  add_link_options(-fsanitize=address,undefined)
endif()

# The capture library (pair-buffer layout, attach, liveness), built from ../capture without changes.
set(CAPTURE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../capture" CACHE PATH "capture source directory")
set(CMAKE_DISABLE_FIND_PACKAGE_realsense2 ON)
add_subdirectory(${CAPTURE_DIR} capture_build EXCLUDE_FROM_ALL)

set(TRK_WARN -Wall -Wextra)
if(TRK_WERROR)
  list(APPEND TRK_WARN -Werror)
endif()

find_package(Threads REQUIRED)

add_library(trk_thirdparty INTERFACE)
target_include_directories(trk_thirdparty SYSTEM INTERFACE third_party/eigen third_party/nanoflann)
target_compile_definitions(trk_thirdparty INTERFACE EIGEN_MPL2_ONLY)

add_library(trk_shm STATIC src/common/futex.cpp src/shmring/shm_ring.cpp)
target_include_directories(trk_shm PUBLIC src)
target_compile_options(trk_shm PRIVATE ${TRK_WARN})
target_link_libraries(trk_shm PUBLIC Threads::Threads)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_link_libraries(trk_shm PUBLIC rt)
endif()

file(GLOB TRK_ICP_SRC CONFIGURE_DEPENDS src/icp/*.cpp)
file(GLOB TRK_CORE_SRC CONFIGURE_DEPENDS src/core/*.cpp src/records/*.cpp src/common/json.cpp)
add_library(trk_core STATIC ${TRK_ICP_SRC} ${TRK_CORE_SRC})
target_include_directories(trk_core PUBLIC src)
target_compile_options(trk_core PRIVATE ${TRK_WARN})
target_link_libraries(trk_core PUBLIC trk_shm trk_thirdparty capture_core)

file(GLOB TRK_TEST_SRC CONFIGURE_DEPENDS tests/test_*.cpp)
add_executable(trk_tests ${TRK_TEST_SRC} tests/alloc_counter.cpp)
target_compile_options(trk_tests PRIVATE ${TRK_WARN})
target_link_libraries(trk_tests PRIVATE trk_core)

file(GLOB TRK_APPS CONFIGURE_DEPENDS src/apps/*.cpp)
# Live programs (signals, threads named with the Linux API) are built on Linux only; the Mac builds tests and tools.
set(TRK_LINUX_ONLY trk_tracker trk_bridge trk_record trk_replay trk_posedump trk_camid trk_calib trk_bgmap)
foreach(app ${TRK_APPS})
  get_filename_component(n ${app} NAME_WE)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" AND n IN_LIST TRK_LINUX_ONLY)
    continue()
  endif()
  add_executable(${n} ${app})
  target_compile_options(${n} PRIVATE ${TRK_WARN})
  target_link_libraries(${n} PRIVATE trk_core)
endforeach()
target_compile_definitions(trk_tests PRIVATE TRK_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
